cmake_minimum_required(VERSION 3.16)
project(ld_architecture_probes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(LD_AUDIT_DEPS "" CACHE PATH "Folder with eigen-5.0.0/Eigen and json.hpp")
if(NOT EXISTS "${LD_AUDIT_DEPS}/eigen-5.0.0/Eigen/Dense" OR NOT EXISTS "${LD_AUDIT_DEPS}/json.hpp")
  message(FATAL_ERROR "Supply exact audit dependencies with -DLD_AUDIT_DEPS=...")
endif()
add_library(native_probe SHARED native_probe.cpp)
add_library(core_probe SHARED core_probe.cpp)
set_target_properties(native_probe core_probe PROPERTIES PREFIX "")
add_executable(report_probe report_probe.cpp core_probe.cpp)
add_executable(sanitizer_probe native_probe.cpp core_probe.cpp)
target_compile_definitions(sanitizer_probe PRIVATE LD_NATIVE_PROBE_MAIN)
foreach(t native_probe core_probe report_probe sanitizer_probe)
  target_include_directories(${t} SYSTEM PRIVATE "${LD_AUDIT_DEPS}/eigen-5.0.0" "${LD_AUDIT_DEPS}")
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(sanitizer_probe PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sanitizer_probe PRIVATE -fsanitize=address,undefined)
endif()
